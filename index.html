""" gerar_escala_html.py ──────────────────── Converte ESCALA_CENTRAL_TELEATENDIMENTO.xlsx em HTML responsivo. Lógica da planilha: - Dias úteis (seg-sex): None = trabalho, 'Feriado' = feriado nacional - Fins de semana (sáb-dom): 'T' = trabalho | 'F' = folga - Turno 'FÉRIAS' = colaborador em férias (mostra banner especial) Cores personalizadas: - FDS trabalhado : #1B263B (azul-escuro) - FDS folga : #FCA311 (âmbar) Requisitos: pip install openpyxl Uso: python gerar_escala_html.py [entrada.xlsx] [saida.html] """ import sys from pathlib import Path from openpyxl import load_workbook from collections import Counter INPUT_FILE = "ESCALA_CENTRAL_TELEATENDIMENTO.xlsx" OUTPUT_FILE = None ROW_TITLE = 2 ROW_HEADERS = 6 ROW_DATES = 7 ROW_DATA_START = 8 COL_FUNCAO = 0 COL_TURNO = 1 COL_NOME = 2 COL_HORARIO = 3 COL_DAYS_START = 4 TURNOS_VALIDOS = {"MATUTINO", "VESPERTINO", "NOTURNO", "MADRUGADA"} TURNO_FERIAS = "FÉRIAS" FINS_DE_SEMANA = {"SÁB", "SAB", "DOM"} # ── Leitura ─────────────────────────────────────────────────────────────────── def ler_excel(caminho): wb = load_workbook(caminho, data_only=True) ws = wb.active rows = list(ws.iter_rows(values_only=True)) titulo = next((str(v) for v in rows[ROW_TITLE-1] if v), "Escala Mensal") dow_row = rows[ROW_HEADERS - 1] date_row = rows[ROW_DATES - 1] dias = [] for i in range(COL_DAYS_START, len(date_row)): num = date_row[i] dow = str(dow_row[i] or "").strip().upper() if isinstance(num, int): dias.append({"col": i, "dia": num, "dow": dow, "fds": dow in FINS_DE_SEMANA}) colaboradores = [] for row in rows[ROW_DATA_START - 1:]: nome = row[COL_NOME] if len(row) > COL_NOME else None turno = row[COL_TURNO] if len(row) > COL_TURNO else None funcao = row[COL_FUNCAO] if len(row) > COL_FUNCAO else None horario = row[COL_HORARIO] if len(row) > COL_HORARIO else None if not nome or not turno: continue nome_s = str(nome).strip() turno_s = str(turno).strip().upper() ferias = (turno_s == TURNO_FERIAS.upper()) if turno_s not in TURNOS_VALIDOS and not ferias: continue escala = {} for d in dias: val = row[d["col"]] if len(row) > d["col"] else None escala[d["dia"]] = str(val).strip() if val is not None else None colaboradores.append({ "nome": nome_s, "funcao": str(funcao or "").strip().title(), "turno": turno_s.capitalize() if not ferias else "Férias", "horario": str(horario or "").strip(), "escala": escala, "ferias": ferias, }) return {"titulo": titulo, "dias": dias, "colaboradores": colaboradores} # ── Card ────────────────────────────────────────────────────────────────────── COR_TURNO = { "Matutino": ("#E6F1FB", "#185FA5"), "Vespertino":("#FEF3CD", "#9A6200"), "Noturno": ("#EEEDFE", "#534AB7"), "Madrugada": ("#EAF3DE", "#3B6D11"), "Férias": ("#FEF9EC", "#8B5E00"), } def iniciais(nome): p = nome.split() return (p[0][0] + p[-1][0]).upper() if len(p) >= 2 else nome[:2].upper() def card_colaborador(col, dias): nome = col["nome"] turno = col["turno"] ferias = col["ferias"] bg, fg = COR_TURNO.get(turno, ("#F1EFE8", "#5F5E5A")) dow_map = {"DOM":0,"SEG":1,"TER":2,"QUA":3,"QUI":4,"SEX":5,"SÁB":6,"SAB":6} grade = [None] * dow_map.get(dias[0]["dow"], 0) + dias while len(grade) % 7: grade.append(None) dias_html = "".join(f'
{c}
' for c in ["D","S","T","Q","Q","S","S"]) ct = {"T": 0, "F": 0} for slot in grade: if slot is None: dias_html += '
' continue val = col["escala"].get(slot["dia"]) # None ou string fds = slot["fds"] val_u = (val or "").upper() if val_u in ("FERIADO",): cls, sty, title = "holiday", "", "Feriado" elif fds: # Fim de semana: T ou F explícito if val_u == "T": cls = "wknd-work" sty = 'style="background:#1B263B;color:#E8EDF2;"' title = "Trabalho — fim de semana" ct["T"] += 1 elif val_u == "F": cls = "wknd-off" sty = 'style="background:#FCA311;color:#4A2800;"' title = "Folga — fim de semana" ct["F"] += 1 else: cls, sty, title = "weekend", "", slot["dow"].capitalize() else: # Dia útil: None = trabalho normal if val is None: cls, sty, title = "work", "", "Trabalho" ct["T"] += 1 elif val_u == "F": cls, sty, title = "off", "", "Folga" ct["F"] += 1 else: cls, sty, title = "weekday", "", f"Dia {slot['dia']}" dias_html += f'
{slot["dia"]}
' ferias_banner = "" ferias_badge = "" if ferias: ferias_banner = '
Colaborador(a) em férias neste período
' ferias_badge = 'Férias' return f"""
{iniciais(nome)}
{nome.title()}
{col['funcao']} · {'—' if ferias else turno} · {'—' if ferias else col['horario']}
{ferias_badge} {ct["T"]}T {ct["F"]}F
{ferias_banner}
{dias_html}
""" # ── HTML ────────────────────────────────────────────────────────────────────── def gerar_html(dados): titulo = dados["titulo"] dias = dados["dias"] cols = dados["colaboradores"] cards_html = "\n".join(card_colaborador(c, dias) for c in cols) total = len(cols) n_ferias = sum(1 for c in cols if c["ferias"]) ct = Counter(c["turno"] for c in cols if not c["ferias"]) turno_tags = "".join(f'{t}: {n}' for t, n in sorted(ct.items())) if n_ferias: turno_tags += f'Férias: {n_ferias}' return f""" {titulo}
{titulo}
{total} colaboradores{turno_tags}
Trabalho (semana)
Trabalho (fim de semana)
Folga (semana)
Folga (fim de semana)
Feriado
Férias
{cards_html}
Nenhum colaborador encontrado para "".
""" # ── Main ────────────────────────────────────────────────────────────────────── def main(): entrada = sys.argv[1] if len(sys.argv) > 1 else INPUT_FILE saida = sys.argv[2] if len(sys.argv) > 2 else OUTPUT_FILE if saida is None: saida = Path(entrada).with_suffix(".html").name if not Path(entrada).exists(): print(f"[ERRO] Arquivo não encontrado: {entrada}") sys.exit(1) print(f"Lendo: {entrada}") dados = ler_excel(entrada) n_ferias = sum(1 for c in dados["colaboradores"] if c["ferias"]) print(f" {len(dados['colaboradores'])} colaboradores | {n_ferias} em férias | {len(dados['dias'])} dias | {dados['titulo']}") print(f"Gerando: {saida}") html = gerar_html(dados) Path(saida).write_text(html, encoding="utf-8") print(f" Concluído! ({Path(saida).stat().st_size // 1024} KB)") print() print("Próximos passos:") print(" 1. Abra o arquivo no navegador para conferir.") print(" 2. Renomeie para index.html e suba no GitHub.") if __name__ == "__main__": main()
